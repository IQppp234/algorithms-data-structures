Задание 2

#include <iostream>
using namespace std;

int main() {
    int n;
    int temp = 1;
    int sumpr = 1;
    cin >> n;

    for(int i = 1;i <= n;i++) {
        temp *= i;
        sumpr *= temp;
    }

    cout << sumpr << endl;
    
    return 0;
}
